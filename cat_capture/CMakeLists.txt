cmake_minimum_required(VERSION 3.8)
project(cat_capture)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/panorama_capture scripts/panorama_rotate DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(rotate_trigger test/test_rotate_trigger.py ENV PYTHONNOUSERSITE=1)
  ament_add_pytest_test(session test/test_session.py ENV PYTHONNOUSERSITE=1)
  ament_add_pytest_test(capture_topic test/test_capture_topic.py ENV PYTHONNOUSERSITE=1)
endif()
ament_package()
